Table users {
  id integer [primary key]
  email varchar
  username varchar
  password varchar
}

Table books {
  id integer [primary key]
  title varchar
  img integer
  category_id integer
  form varchar
  isbn varchar
  summary varchar
  detail text
  author varchar
  pages integer
  content text
  price integer
  pub_date date
}

Table category {
  category_id integer [primary key]
  category_name varchar
}

Table likes {
  user_id integer
  liked_book_id integer
}

Table cartItems {
  id integer
  book_id integer
  quantity integer
  user_id integer
}

Ref: books.category_id > category.id

Ref: likes.user_id > users.id
Ref: likes.liked_book_id > books.id

Ref: cartItems.book_id > books.id
Ref: cartItems.user_id > users.id
