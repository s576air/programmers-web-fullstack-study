Table users {
  id integer [primary key]
  email varchar
  username varchar
  password varchar
}

Table books {
  id integer [primary key]
  title varchar
  img integer
  category_id integer
  form varchar
  isbn varchar
  summary varchar
  detail text
  author varchar
  pages integer
  content text
  price integer
  pub_date date
}

Table category {
  category_id integer [primary key]
  category_name varchar
}

Table likes {
  user_id integer
  liked_book_id integer
}

Table cartItems {
  id integer
  book_id integer
  quantity integer
  user_id integer
}

Table orders {
  id integer [primary key]
  delivery_id integer
  user_id integer
  book_title varchar
  total_quantity integer
  created_at timestamp
}

Table orderedBook {
  id integer [primary key]
  order_id integer
  book_id integer
  quantity integer
}

Table delivery {
  id integer [primary key]
  order_id varchar
  receiver varchar
  content varchar
}

Ref: books.category_id > category.id

Ref: likes.user_id > users.id
Ref: likes.liked_book_id > books.id

Ref: cartItems.book_id > books.id
Ref: cartItems.user_id > users.id

Ref: orders.delivery_id > delivery.id
Ref: orders.user_id > users.id

Ref: orderedBook.order_id > orders.id
Ref: orderedBook.book_id > books.id

Ref: delivery.order_id > orders.id
